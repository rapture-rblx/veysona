# A one-ZIP source release contains every WAV losslessly packed as FLAC, rather
# than three separate ZIPs. Rebuild the original, SHA256-verified WAV data
# automatically before source tests / JUCE resource copying.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/packed_soundbank/sha256_manifest.tsv")
 if(WIN32)
  find_program(VEYSONA_POWERSHELL_EXE NAMES powershell.exe pwsh REQUIRED)
  execute_process(COMMAND "${VEYSONA_POWERSHELL_EXE}" -NoProfile -NonInteractive -ExecutionPolicy Bypass
     -File "${CMAKE_CURRENT_SOURCE_DIR}/scripts/restore_sound_bank.ps1"
   RESULT_VARIABLE VEYSONA_SOUND_BANK_RESTORE_STATUS)
 else()
  execute_process(COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/scripts/restore_sound_bank.sh"
   RESULT_VARIABLE VEYSONA_SOUND_BANK_RESTORE_STATUS)
 endif()
 if(NOT VEYSONA_SOUND_BANK_RESTORE_STATUS EQUAL 0)
  message(FATAL_ERROR "Could not restore and verify Veysona's bundled WAV sound bank")
 endif()
endif()

# Release channel configuration is embedded at build time. Keep the signing thumbprint
# out of user-editable preference files; never silently run unsigned installers.
set(VEYSONA_UPDATE_MANIFEST_URL "" CACHE STRING "HTTPS URL of Veysona's published updates/stable.json manifest")
set(VEYSONA_UPDATE_SIGNER_THUMBPRINT "" CACHE STRING "40-character Windows Authenticode publisher certificate thumbprint")
if(VEYSONA_UPDATE_MANIFEST_URL AND NOT VEYSONA_UPDATE_MANIFEST_URL MATCHES "^https://[A-Za-z0-9.-]+/[^; ]+$")
 message(FATAL_ERROR "VEYSONA_UPDATE_MANIFEST_URL must be a public HTTPS release manifest URL")
endif()
if(VEYSONA_UPDATE_SIGNER_THUMBPRINT AND NOT VEYSONA_UPDATE_SIGNER_THUMBPRINT MATCHES "^[A-Fa-f0-9]+$")
 message(FATAL_ERROR "The publisher thumbprint must contain hexadecimal characters only")
endif()
if(VEYSONA_UPDATE_SIGNER_THUMBPRINT)
 # Exact length validation is also enforced by the runtime update policy.
 string(LENGTH "${VEYSONA_UPDATE_SIGNER_THUMBPRINT}" VEYSONA_SIGNER_LENGTH)
 if(NOT VEYSONA_SIGNER_LENGTH EQUAL 40)
  message(FATAL_ERROR "Publisher thumbprint must contain exactly 40 hexadecimal characters")
 endif()
endif()
